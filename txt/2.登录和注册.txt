添加驱动文件：
方法 1：

b. 将 libmysql.dll 放入 Qt 安装路径下的 5.14.2/mingw73_64/bin 目录下（如果不行，替换或你自己目录下的这个文件：C:\Program Files\MySQL\MySQL Server 5.7\bin）

void OperateDb::connect()
{

    m_db.setPassword("123456");
    m_db.setDatabaseName("mydb");

        qDebug() << "数据库连接成功";
    }
    else
    {
        qDebug() << "数据库连接失败" << m_db.lastError().text();
    }
}

OperateDb& OperateDb::getInstance()
{
    static OperateDb instance;
    return instance;
}

注册功能

1.客户端给服务器发送请求（用户名和密码）
a. 注册按钮转到槽
b. 通过 ui 的输入框获取用户名和密码，判断用户名和密码的长度是否为空或超过 32
c. 构建 pdu，用户名和密码放到 caData
d. 协议的枚举值中添加注册请求和响应类型，并将请求类型给 pdu
e. 通过 socket 讲 pdu 发送给服务器

2.服务器接收并取出数据、处理注册、结果响应给客户端
a. 接收函数中判断类型为注册的请求，进行处理
b. 从 pdu 的 caData 中取出用户名和密码
c. 数据库操作类定义一个函数去处理注册并调用，参数为用户名和密码，返回值为 bool
i. 先对指针参数判空
ii. 构建一个查询用户是否存在的 sql 语句，执行并判断是否有结果
iii. 构建一个插入新用户的 sql 语句，执行并返回结果
d. 得到注册结果，构建一个响应 pdu，将注册结果放入 caData，消息类型赋值为注册的响应
e. 通过 socket 发送 pdu 给客户端

3.客户端接收并取出结果，展示结果给用户
a. 接收消息函数中判断类型为注册的响应，进行处理
b. 定义一个处理结果的变量，并从 caData 中取出
c. 根据处理结果弹窗展示给用户


void Client::on_regist_PB_clicked()
{
    QString strName = ui->name_LE->text();
    QString strPwd = ui->pwd_LE->text();
    if (strName.isEmpty() || strPwd.isEmpty() ||
        strName.toStdString().size() > 32 ||
        strPwd.toStdString().size() > 32)
    {
        QMessageBox::information(this, "提示", "用户名或密码长度非法");
        return;
    }
    PDU* pdu = mkPDU(0);
    memcpy(pdu->caData, strName.toStdString().c_str(), 32);
    memcpy(pdu->caData+32, strPwd.toStdString().c_str(), 32);
    pdu->uiMsgType = ENUM_MSG_TYPE_REGIST_REQUEST;
    socket.write((char*)pdu, pdu->uiTotalLen);
    qDebug() << "send msg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    free(pdu);
    pdu = NULL;
}


void MyTcpSocket::recvMsg()
{
    qDebug() << "recvMsg 接收消息长度" << this->bytesAvailable();
    //先读取 pdu 总长度这个成员变量
    uint uiPDULen = 0;
    this->read((char*)&uiPDULen, sizeof(uint));
    //根据总长度构建 pdu，读取结构体中剩余的成员变量
    uint uiMsgLen = uiPDULen - sizeof(PDU);
    PDU* pdu = mkPDU(uiMsgLen);
    this->read((char*)pdu+sizeof(uint), uiPDULen-sizeof(uint));
    qDebug() << "recvMsg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    //处理消息
    switch (pdu->uiMsgType)
    {
    case ENUM_MSG_TYPE_REGIST_REQUEST:
    {
        char caName[32] = {'\0'};
        char caPwd[32] = {'\0'};
        memcpy(caName, pdu->caData, 32);
        memcpy(caPwd, pdu->caData+32, 32);
        bool ret = OperateDb::getInstance().handleRegist(caName, caPwd);
        qDebug() << "regist ret" << ret;
        PDU* respdu = mkPDU(0);
        memcpy(respdu->caData, &ret, sizeof(bool));
        respdu->uiMsgType = ENUM_MSG_TYPE_REGIST_RESPOND;
        this->write((char*)respdu, respdu->uiTotalLen);
        qDebug() << "send msg respdu->uiTotalLen" << respdu->uiTotalLen
                 << "respdu->uiMsgLen" << respdu->uiMsgLen
                 << "respdu->uiMsgType" << respdu->uiMsgType
                 << "respdu->caData" << respdu->caData
                 << "respdu->caData+32" << respdu->caData+32
                 << "respdu->caMsg" << respdu->caMsg;
        free(respdu);
        respdu = NULL;
        break;
    }
    default:
        break;
    }
    free(pdu);
    pdu = NULL;
}


bool OperateDb::handleRegist(char *caName, char *caPwd)
{
    if (caName == NULL || caPwd == NULL)
    {
        return false;
    }
    QString sql = QString("select * from user_info where name='%1'").arg(caName);
    qDebug() << "查询用户是否存在 sql" << sql;
    QSqlQuery q;
    if (!q.exec(sql))
    {
        return false;
    }
    if (q.next())
    {
        return false;
    }
    sql = QString("insert into user_info(name, pwd) values('%1', '%2')").arg(caName).arg(caPwd);
    qDebug() << "插入一条新用户 sql" << sql;
    return q.exec(sql);
}


void Client::recvMsg()
{
    qDebug() << "recvMsg 接收消息长度" << socket.bytesAvailable();
    //先读取 pdu 总长度这个成员变量
    uint uiPDULen = 0;
    socket.read((char*)&uiPDULen, sizeof(uint));
    //根据总长度构建 pdu，读取结构体中剩余的成员变量
    uint uiMsgLen = uiPDULen - sizeof(PDU);
    PDU* pdu = mkPDU(uiMsgLen);
    socket.read((char*)pdu+sizeof(uint), uiPDULen-sizeof(uint));
    qDebug() << "recvMsg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    //处理消息
    switch (pdu->uiMsgType)
    {
    case ENUM_MSG_TYPE_REGIST_RESPOND:
    {
        bool ret;
        memcpy(&ret, pdu->caData, sizeof(bool));
        if (ret)
        {
            QMessageBox::information(this, "提示", "注册成功");
        }
        else
        {
            QMessageBox::information(this, "提示", "注册失败");
        }
        break;
    }
    default:
        break;
    }
    free(pdu);
    pdu = NULL;
}

登录功能

1.客户端发送请求给服务器
a. 登录按钮转到槽函数
b. 通过 ui 获取输入框中的用户名和密码
c. 构建 pdu，将用户名和密码放到 caData 里，协议中新增登录的请求和响应类型并赋值
d. 定义 m_strLoginName 成员变量并赋值
e. 通过 socket 发送 pdu（打印和释放）

2.服务器接收并处理请求，结果响应给客户端
a. MyTcpSocket 的接收消息函数 switch 加一个登录请求的 case
b. 将用户名和密码从 caData 中取出
c. 数据库操作类中新增一个处理登录的函数，参数为用户名和密码，返回值 bool 类型
i. 构建 sql 语句判断该用户名和密码是否存在
ii. 如果存在，构建 sql 语句将 online 字段置为 1
d. 调用处理登录函数，得到处理结果
e. 定义 m_strLoginName 成员变量，如果处理成功赋值
f. 构建响应 pdu，将结果放入 caData，赋值响应类型
g. 发送响应 pdu（打印和释放）

3.客户端根据结果展示弹窗
a. 接收消息函数中增加一个登录响应的 case
b. 将处理结果从 caData 中取出，并根据结果弹窗显示


bool OperateDb::handleLogin(const char *caName, char *caPwd)
{
    if (caName == NULL || caPwd == NULL)
    {
        return false;
    }
    QString sql = QString("select * from user_info where name='%1' and pwd='%2'").arg(caName).arg(caPwd);
    qDebug() << "查询用户是否存在 sql" << sql;
    QSqlQuery q;
    if (!q.exec(sql))
    {
        return false;
    }
    if (!q.next())
    {
        return false;
    }
    sql = QString("update user_info set online=1 where name='%1' and pwd='%2'").arg(caName).arg(caPwd);
    qDebug() << "更新用户 online 为 1 sql" << sql;
    return q.exec(sql);
}

void OperateDb::handleOffline(const char *caName)
{
    if (caName == NULL)
    {
        return;
    }
    QString sql = QString("update user_info set online=0 where name='%1'").arg(caName);
    qDebug() << "更新用户 online 为 0 sql" << sql;
    QSqlQuery q;
    q.exec(sql);
}


MyTcpSocket::MyTcpSocket()
{
    connect(this, &QTcpSocket::readyRead, this, &MyTcpSocket::recvMsg);
    connect(this, &QTcpSocket::disconnected, this, &MyTcpSocket::clientOffline);
}

void MyTcpSocket::clientOffline()
{
    OperateDb::getInstance().handleOffline(m_strLoginName.toStdString().c_str());
    MyTcpServer::getInstance().removeSocket(this);
}

void MyTcpServer::removeSocket(MyTcpSocket *mySocket)
{
    m_tcpSocketList.removeOne(mySocket);
    mySocket->deleteLater();
    mySocket = NULL;
    for (int i = 0; i < m_tcpSocketList.size(); i++)
    {
        qDebug() << m_tcpSocketList[i]->m_strLoginName;
    }
}
