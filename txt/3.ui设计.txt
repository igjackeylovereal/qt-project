用户功能
查找用户
客户端发送给服务器需要查找的用户名
a. 查找用户按钮转到槽，创建带输入框的弹窗，获取到需要查找的用户名
b. 构建 pdu，用户名放入 caData，新建查找用户的两个消息类型（协议的枚举值中）并赋值
c. 通过 socket 发送给服务器，打印并释放
服务器接收并查找用户的状态并响应给客户端
a. 接收函数中新建一个 case 处理查找用户的请求类型
b. 从 caData 取出用户名
c. 数据库操作类中新建一个处理查找用户的函数，参数为用户名，返回值为 int 类型
d. 构建 sql 语句并执行查询该用户是否在线以及是否在线
e. 调用该数据库函数，得到用户状态
f. 构建 respdu，将用户状态放到 caData，类型赋值，发送给客户端
客户端接收并弹窗展示用户状态
a. 接收函数中新建一个 case 处理响应
b. 将用户状态从 caData 取出
c. 根据状态弹窗显示


void Friend::on_findUser_PB_clicked()
{
    QString strName = QInputDialog::getText(this, "查找用户", "用户名:");
    qDebug() << "strName" << strName;
    if (strName.isEmpty())
    {
        return;
    }
    PDU* pdu = mkPDU(0);
    memcpy(pdu->caData, strName.toStdString().c_str(), 32);
    pdu->uiMsgType = ENUM_MSG_TYPE_FIND_USER_REQUEST;
    Client::getInstance().socket.write((char*)pdu, pdu->uiTotalLen);
    qDebug() << "send msg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    free(pdu);
    pdu = NULL;
}


int OperateDb::handleFindUser(const char *caName)
{
    if (caName == NULL)
    {
        return -1;
    }
    QString sql = QString("select online from user_info where name='%1'").arg(caName);
    qDebug() << "查询用户 online 字段 sql" << sql;
    QSqlQuery q;
    q.exec(sql);
    if (q.next())
    {
        return q.value(0).toInt();
    }
    return 2;
}

void Friend::on_onlineUser_PB_clicked()
{
    if (m_pOnlineUser->isHidden())
    {
        m_pOnlineUser->show();
    }
    PDU* pdu = mkPDU(0);
    pdu->uiMsgType = ENUM_MSG_TYPE_ONLINE_USER_REQUEST;
    Client::getInstance().sendMsg(pdu);
}


void Client::sendMsg(PDU *pdu)
{
    socket.write((char*)pdu, pdu->uiTotalLen);
    qDebug() << "send msg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    free(pdu);
    pdu = NULL;
}


QStringList OperateDb::handleOnlineUser()
{
    QString sql = QString("select name from user_info where online=1");
    QSqlQuery q;
    q.exec(sql);
    QStringList result;
    result.clear();
    while (q.next())
    {
        result.append(q.value(0).toString());
    }
    return result;
}


case ENUM_MSG_TYPE_ONLINE_USER_REQUEST:

}
