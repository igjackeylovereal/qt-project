服务器
MyTcpSocket 的 recvMsg 函数，没有参数，返回值是 PDU*，将读取部分代码封装成函数
处理封装成函数，参数是 PDU，返回值也是 PDU
新建一个类 MsgHandler，新建一个 PDU* 类型的成员变量
MyTcpSocket 类定义 MsgHandler 指针类型的成员变量，在构造函数中 new 出来
MsgHandler 创建每一个请求的处理函数，返回值为 PDU*
响应封装成函数，参数是参数列表为空


class MyTcpSocket : public QTcpSocket
{
    Q_OBJECT

public:
    MyTcpSocket();
    ~MyTcpSocket();
    QString m_strLoginName;
    MsgHandler* m_pMH; //新增成员变量
    void sendMsg(PDU* pdu);
    PDU* readMsg();
    void recvMsg();

public slots:
    void clientOffline();
};

MyTcpSocket::MyTcpSocket()
{
    connect(this, &QTcpSocket::readyRead, this, &MyTcpSocket::recvMsg);
    connect(this, &QTcpSocket::disconnected, this, &MyTcpSocket::clientOffline);
    m_pMH = new MsgHandler;
}

MyTcpSocket::~MyTcpSocket()
{
    delete m_pMH;
}

void MyTcpSocket::sendMsg(PDU* pdu)
{
    if (pdu == NULL)
    {
        return;
    }
    this->write((char*)pdu, pdu->uiTotalLen);
    qDebug() << "send msg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    free(pdu);
    pdu = NULL;
}

PDU* MyTcpSocket::readMsg()
{
    qDebug() << "recvMsg 接收消息长度" << this->bytesAvailable();
    //先读取 pdu 总长度这个成员变量
    uint uiPDULen = 0;
    this->read((char*)&uiPDULen, sizeof(uint));
    //根据总长度构建 pdu，读取结构体中剩余的成员变量
    uint uiMsgLen = uiPDULen - sizeof(PDU);
    PDU* pdu = mkPDU(uiMsgLen);
    this->read((char*)pdu+sizeof(uint), uiPDULen-sizeof(uint));
    qDebug() << "recvMsg pdu->uiTotalLen" << pdu->uiTotalLen
             << "pdu->uiMsgLen" << pdu->uiMsgLen
             << "pdu->uiMsgType" << pdu->uiMsgType
             << "pdu->caData" << pdu->caData
             << "pdu->caData+32" << pdu->caData+32
             << "pdu->caMsg" << pdu->caMsg;
    return pdu;
}

PDU* MyTcpSocket::handleMsg(PDU* pdu)
{
    PDU* respdu = NULL;
    //处理消息
    switch (pdu->uiMsgType)
    {
    case ENUM_MSG_TYPE_REGIST_REQUEST:
        respdu = m_pMH->regist();
        break;
    case ENUM_MSG_TYPE_LOGIN_REQUEST:
        respdu = m_pMH->login(m_strLoginName);
        break;
    case ENUM_MSG_TYPE_FIND_USER_REQUEST:
        respdu = m_pMH->findUser();
        break;
    case ENUM_MSG_TYPE_ONLINE_USER_REQUEST:
        respdu = m_pMH->onlineUser();
        break;
    default:
        break;
    }
    free(pdu);
    pdu = NULL;
    return respdu;
}

void MyTcpSocket::recvMsg()
{
    PDU* pdu = readMsg();
    PDU* respdu = handleMsg(pdu);
    sendMsg(respdu);
}

public:

PDU* MsgHandler::regist()
{
    char caName[32] = {'\0'};
    char caPwd[32] = {'\0'};
    memcpy(caName, pdu->caData, 32);
    memcpy(caPwd, pdu->caData+32, 32);
    bool ret = OperateDb::getInstance().handleRegist(caName, caPwd);
    qDebug() << "regist ret" << ret;
    PDU* respdu = mkPDU(0);
    memcpy(respdu->caData, &ret, sizeof(bool));
    respdu->uiMsgType = ENUM_MSG_TYPE_REGIST_RESPOND;
    return respdu;
}

刷新在线好友
客户端给服务器发送当前登录的用户名
用户名放入 pdu 的 caData 发送给服务器
服务器接收并处理请求，获取所有在线好友名，发送给客户端
数据库操作类定义一个处理在线好友的函数，参数为用户名，返回值 QStringList
执行通过用户名获取在线好友的 sql 语句
q.next() 循环将结果，将好友的用户名放入 QStringList 并返回
调用数据库处理函数，得到 QStringList 在线好友名
构建 respdu，caMsg 为好友名数量*32
遍历在线好友名，将好友名放入 caMsg
响应给客户端
客户端接收在线好友名放入好友列表框中
Friend 类定义更新好友列表框函数，参数为 QStringList，先清空再添加到列表框
好友数量为 pdu 中的 uiMsgLen/32
遍历 caMsg 中的好友名，将其放入 QStringList 并调用更新好友列表框函数
刷新好友封装成函数，在 Friend 的构造函数和添加好友成功时调用


QStringList OperateDb::handleFlushFriend(char *caName)
{
    QStringList res;
    if (caName == NULL)
    {
        return res;
    }
    QString sql = QString("select name from user_info "
                          "where id in ("
                          "    select user_id from friend where friend_id="
                          "    (select id from user_info where name='%1')"
                          "    union"
                          "    select friend_id from friend where user_id="
                          "    (select id from user_info where name='%1')"
                          ") and online=1").arg(caName);
    QSqlQuery q;
    q.exec(sql);
    while (q.next())
    {
        res.append(q.value(0).toString());
    }
    return res;
}


PDU* MsgHandler::flushFriend()
{
    char caName[32] = {'\0'};
    memcpy(caName, pdu->caData, 32);
    QStringList res = OperateDb::getInstance().handleFlushFriend(caName);
    PDU* respdu = mkPDU(res.size()*32);
    respdu->uiMsgType = ENUM_MSG_TYPE_FLUSH_FRIEND_RESPOND;
    for (int i=0; i<<res.size(); i++)
    {
        memcpy(respdu->caMsg+i*32, res[i].toStdString().c_str(), 32);
    }
    return respdu;
}
